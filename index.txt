<script>
const SHEET_API_URL = "https://script.google.com/macros/s/AKfycbzBBm-upwM2lnIbqW5tIQwnkUCSLk89if_MpV6lJ_pBwL0ciagR11LPnDVdGOhFCPnD7g/exec";

let itemCount = 0;
function addItem() {
  itemCount++;
  const container = document.getElementById('itemsContainer');
  const row = document.createElement('div');
  row.className = 'row';
  row.innerHTML = `
    <input type="text" class="name" placeholder="ชื่อรายการ (เช่น ข้าวผัด)">
    <input type="number" class="price" placeholder="ราคารวม">
    <button class="remove-btn" onclick="this.parentElement.remove()">✕</button>
  `;
  container.appendChild(row);
}
addItem();

function generateReceipt() {
  const names = document.querySelectorAll('#itemsContainer .name');
  const prices = document.querySelectorAll('#itemsContainer .price');
  const people = parseInt(document.getElementById('peopleCount').value) || 1;

  let items = [];
  let total = 0;
  for (let i = 0; i < names.length; i++) {
    const name = names[i].value.trim();
    const price = parseFloat(prices[i].value) || 0;
    if (name && price > 0) {
      items.push({ name, price });
      total += price;
    }
  }
  if (items.length === 0) { alert('กรุณาใส่รายการอย่างน้อย 1 รายการ'); return; }

  const now = new Date();
  const dateStr = now.toLocaleDateString('th-TH', {day:'2-digit',month:'2-digit',year:'numeric'}) + ' ' +
                  now.toLocaleTimeString('th-TH', {hour:'2-digit',minute:'2-digit'});
  const billCode = 'CA' + now.getFullYear() + String(now.getMonth()+1).padStart(2,'0') +
                    String(now.getDate()).padStart(2,'0') + '-' +
                    String(now.getHours()).padStart(2,'0') + String(now.getMinutes()).padStart(2,'0');

  document.getElementById('billCode').textContent = billCode;
  document.getElementById('billDate').textContent = dateStr;

  const itemsList = document.getElementById('itemsList');
  itemsList.innerHTML = '';
  items.forEach(it => {
    const row = document.createElement('div');
    row.className = 'item-row';
    row.innerHTML = `<span class="name">${it.name}</span><span class="price">${it.price.toLocaleString('th-TH',{minimumFractionDigits:2})}</span>`;
    itemsList.appendChild(row);
  });

  document.getElementById('totalCount').textContent = items.length;
  document.getElementById('totalSum').textContent = total.toLocaleString('th-TH',{minimumFractionDigits:2});
  document.getElementById('grandTotal').textContent = total.toLocaleString('th-TH',{minimumFractionDigits:2}) + ' บาท';
  document.getElementById('splitLabel').textContent = `จ่ายเฉลี่ย (${people} คน)`;
  document.getElementById('splitAmount').textContent = (total/people).toLocaleString('th-TH',{minimumFractionDigits:2}) + ' บ.';

  document.getElementById('formCard').style.display = 'none';
  document.getElementById('receipt').style.display = 'block';
  document.getElementById('actionBtns').style.display = 'flex';

  saveToSheet({ billCode, date: dateStr, items, total, people, perPerson: total/people });
}

function saveToSheet(data) {
  fetch(SHEET_API_URL, {
    method: 'POST',
    mode: 'no-cors',
    headers: { 'Content-Type': 'text/plain' },
    body: JSON.stringify(data)
  }).catch(err => console.log('บันทึกลง Sheet ไม่สำเร็จ', err));
}

function saveAsImage() {
  const receipt = document.getElementById('receipt');
  html2canvas(receipt, { scale: 2, backgroundColor: '#ffffff' }).then(canvas => {
    const link = document.createElement('a');
    link.download = 'receipt-' + Date.now() + '.png';
    link.href = canvas.toDataURL('image/png');
    link.click();
  });
}

if ('serviceWorker' in navigator) {
  window.addEventListener('load', () => {
    navigator.serviceWorker.register('sw.js');
  });
}
